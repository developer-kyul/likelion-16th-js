재귀함수

```jsx
attr(document.getElementById("app"), {
  class: "className",
  title: "hey",
});

function attr(el, prop, value) {
  // 1. prop 매개변수의 타입이 객체인가요? (조건)
  //    조건 A : el의 타입이 'object'인가요? -> null, array, object 중 하나
  //    조건 B : el은 null이 아닌가요? -> array, object 중 하나
  //    조건 C : el은 배열이 아닌가요? -> object만 남음!
  const isObject = typeof prop === "object" && prop && !Array.isArray(prop);
  // console.log(isObject)

  //    조건 A 그리고 조건 B 그리고 조건 C 모두 참인 경우
  // 2. prop 타입이 객체라면? (조건이 참)
  if (isObject) {
    // 3. 객체의 키:값 쌍을 순환 (for...in 문 활용)
    for (const propName in prop) {
      const propValue = prop[propName];
      // console.log(el, key, ' - ', value)
      // [0] el에 key(id) 속성 값으로 value(idName)를 설정
      // [1] el에 key(class) 속성 값으로 value(className)를 설정
      // [2] el에 key(title) 속성 값으로 value(titleName)를 설정
      // ...
      // 4. 재귀 호출 (attr 함수 내부에서 attr 함수를 호출)
      attr(el, propName, propValue);
    }

    // 함수 종료 (객체 순환이 마무리 되면, 안 그러면 무한 루프...)
    console.log("함수 종료");
    return;
  }

  if (value === undefined) {
    return el.getAttribute(prop);
  }

  if (value === null) {
    return el.removeAttribute(prop);
  }

  el.setAttribute(prop, value);
}
```
